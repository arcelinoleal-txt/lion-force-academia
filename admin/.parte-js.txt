function saveJSON(path, obj, message) {
            var body = JSON.stringify(obj, null, 2) + '\n';
            return apiGet(path).then(function (j) {
                return fetch(API + path, {
                    method: 'PUT', headers: Object.assign({ 'Content-Type': 'application/json' }, headers()),
                    body: JSON.stringify({ message: message, content: b64encode(body), sha: j.sha, branch: BRANCH })
                });
            }).then(function (r) { if (!r.ok) throw new Error('HTTP ' + r.status); return r.json(); });
        }

function uploadImage(file, input, previewImg) {
            var reader = new FileReader();
            reader.onload = function () {
                var base64 = String(reader.result).split(',')[1];
                var ext = (file.name.split('.').pop() || 'jpg').toLowerCase().slice(0, 4);
                var name = 'fotos/' + slug(file.name.replace(/\.[^.]+$/, '')) + '-' + Date.now() + '.' + ext;
                input.value = 'Carregando...';
                fetch(API + name, {
                    method: 'PUT', headers: Object.assign({ 'Content-Type': 'application/json' }, headers()),
                    body: JSON.stringify({ message: 'Admin: envia ' + name, content: base64, branch: BRANCH })
                }).then(function (r) { if (!r.ok) throw new Error('HTTP ' + r.status); return r.json(); })
                .then(function () { input.value = name; if (previewImg) { previewImg.src = name.startsWith('http') ? name : '../' + name; } msg(input.dataset.msg || 'galMsg', 'Foto enviada!', true); })
                .catch(function () { input.value = ''; msg(input.dataset.msg || 'galMsg', 'Falha no envio. Confira o token.', false); });
            };
            reader.readAsDataURL(file);
        }

function sha256(str) {
            return crypto.subtle.digest('SHA-256', new TextEncoder().encode(str)).then(function (buf) {
                return Array.prototype.map.call(new Uint8Array(buf), function (b) { return ('0' + b.toString(16)).slice(-2); }).join('');
            });
        }

        /* Cria um cofre novo: mesmo formato do unlockVault, usado ao trocar a senha */
        function lockVault(plainToken, chave) {
            var salt = crypto.getRandomValues(new Uint8Array(16));
            var iv = crypto.getRandomValues(new Uint8Array(16));
            return crypto.subtle.importKey('raw', new TextEncoder().encode(chave), 'PBKDF2', false, ['deriveBits'])
                .then(function (base) {
                    return crypto.subtle.deriveBits({ name: 'PBKDF2', salt: salt, iterations: 120000, hash: 'SHA-1' }, base, 512);
                }).then(function (bits) {
                    var raw = new Uint8Array(bits);
                    var aesKey = raw.slice(0, 32), hmacKey = raw.slice(32, 64);
                    return crypto.subtle.importKey('raw', aesKey, { name: 'AES-CBC', iv: iv }, false, ['encrypt'])
                        .then(function (ak) { return crypto.subtle.encrypt({ name: 'AES-CBC', iv: iv }, ak, new TextEncoder().encode(plainToken)); })
                        .then(function (data) {
                            var cat = new Uint8Array(salt.length + iv.length + data.length);
                            cat.set(salt, 0); cat.set(iv, salt.length); cat.set(data, salt.length + iv.length);
                            return crypto.subtle.importKey('raw', hmacKey, { name: 'HMAC', hash: 'SHA-256' }, false, ['sign'])
                                .then(function (hk) { return crypto.subtle.sign('HMAC', hk, cat); })
                                .then(function (mac) {
                                    return {
                                        data: bytesToB64(new Uint8Array(data)),
                                        mac: bytesToB64(new Uint8Array(mac)),
                                        iv: bytesToB64(iv),
                                        salt: bytesToB64(salt)
                                    };
                                });
                        });
                });
        }

function unlockVault(password) {
            var salt = b64ToBytes(VAULT.salt), iv = b64ToBytes(VAULT.iv),
                data = b64ToBytes(VAULT.data), mac = b64ToBytes(VAULT.mac);
            return crypto.subtle.importKey('raw', new TextEncoder().encode(password), 'PBKDF2', false, ['deriveBits']).then(function (base) {
                return crypto.subtle.deriveBits({ name: 'PBKDF2', salt: salt, iterations: 120000, hash: 'SHA-1' }, base, 512);
            }).then(function (bits) {
                var raw = new Uint8Array(bits);
                var aesKey = raw.slice(0, 32), hmacKey = raw.slice(32, 64);
                return crypto.subtle.importKey('raw', hmacKey, { name: 'HMAC', hash: 'SHA-256' }, false, ['verify']).then(function (hk) {
                    var cat = new Uint8Array(salt.length + iv.length + data.length);
                    cat.set(salt, 0); cat.set(iv, salt.length); cat.set(data, salt.length + iv.length);
                    return crypto.subtle.verify('HMAC', hk, mac, cat).then(function (ok) {
                        if (!ok) throw new Error('hmac');
                        return crypto.subtle.importKey('raw', aesKey, { name: 'AES-CBC', iv: iv }, false, ['decrypt']);
                    }).then(function (ak) {
                        return crypto.subtle.decrypt({ name: 'AES-CBC', iv: iv }, ak, data);
                    }).then(function (pt) { return new TextDecoder().decode(pt); });
                });
            });
        }

function histField(boxId, obj, key, label, ph, multiline) {
            var l = document.createElement('label'); l.className = 'f'; l.textContent = label;
            var inp = document.createElement(multiline ? 'textarea' : 'input');
            if (!multiline) inp.type = 'text';
            inp.value = obj[key] || ''; inp.placeholder = ph || '';
            inp.addEventListener('input', function () { obj[key] = inp.value; });
            l.appendChild(inp); document.getElementById(boxId).appendChild(l);
            return inp;
        }
